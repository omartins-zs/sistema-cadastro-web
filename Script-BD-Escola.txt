CREATE DATABASE IF NOT EXISTS escola CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE escola;

CREATE TABLE IF NOT EXISTS `usuarios` (
    `id` INT NOT NULL AUTO_INCREMENT,
    `usuario` VARCHAR(120) NOT NULL,
    `senha` VARCHAR(255) NOT NULL,
    `email` VARCHAR(255) NOT NULL,
    `nivel` VARCHAR(120) NOT NULL,
    PRIMARY KEY (`id`),
    UNIQUE (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `boletim` (
    `id` INT NOT NULL AUTO_INCREMENT,
    `usuario_id` INT NOT NULL,
    `disciplina` VARCHAR(100) NOT NULL,
    `nota_b1` DECIMAL(4,1) DEFAULT 0.0,
    `nota_b2` DECIMAL(4,1) DEFAULT 0.0,
    `nota_b3` DECIMAL(4,1) DEFAULT 0.0,
    `nota_b4` DECIMAL(4,1) DEFAULT 0.0,
    `frequencia` INT DEFAULT 100,
    `status` VARCHAR(50) DEFAULT 'Em Curso',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uq_boletim_aluno_disciplina` (`usuario_id`, `disciplina`),
    FOREIGN KEY (`usuario_id`) REFERENCES `usuarios`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `solicitacoes_secretaria` (
    `id` INT NOT NULL AUTO_INCREMENT,
    `usuario_id` INT NOT NULL,
    `tipo_documento` VARCHAR(150) NOT NULL,
    `protocolo` VARCHAR(50) NOT NULL,
    `status` VARCHAR(50) NOT NULL,
    `data_solicitacao` DATETIME NOT NULL,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`usuario_id`) REFERENCES `usuarios`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `comunicados` (
    `id` INT NOT NULL AUTO_INCREMENT,
    `autor_id` INT NOT NULL,
    `titulo` VARCHAR(150) NOT NULL,
    `mensagem` TEXT NOT NULL,
    `criado_em` DATETIME NOT NULL,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`autor_id`) REFERENCES `usuarios`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `mensalidades` (
    `id` INT NOT NULL AUTO_INCREMENT,
    `usuario_id` INT NOT NULL,
    `competencia` VARCHAR(50) NOT NULL,
    `mes_numero` INT NOT NULL,
    `ano` INT NOT NULL DEFAULT 2026,
    `valor` DECIMAL(10,2) NOT NULL DEFAULT 450.00,
    `vencimento` DATE NOT NULL,
    `status` ENUM('Pago', 'Pendente') NOT NULL DEFAULT 'Pendente',
    `data_pagamento` DATETIME NULL,
    `forma_pagamento` VARCHAR(50) NULL,
    `codigo_pix` VARCHAR(255) NULL,
    `linha_digitavel` VARCHAR(100) NULL,
    PRIMARY KEY (`id`),
    KEY `idx_mensalidade_aluno` (`usuario_id`),
    FOREIGN KEY (`usuario_id`) REFERENCES `usuarios`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
